router.post("/save_project", uploadProjectImage([
    { name: "project_image", maxCount: 1 },
    { name: "image", maxCount: 1 },]), async (req, res) => {

  const { project_name, project_description } = req.body;

  const projectImage = req.files?.project_image?.[0];
  const image = req.files?.image?.[0];

  const projectImageData = await uploadToS3(projectImage, projectImage.originalname);
  const imageData = await uploadToS3(image, image.originalname);

  var sql = `INSERT INTO projects (project_name, project_description, project_image, image) VALUES ('${project_name}', '${project_description}', '${projectImageData.fileKey}', '${imageData.fileKey}')`;

  res.send(sql);
});
